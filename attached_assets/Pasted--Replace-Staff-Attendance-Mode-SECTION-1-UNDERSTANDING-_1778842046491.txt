**4.5 — New Helper: `groupDatesByMonth()`**

```javascript
function groupDatesByMonth(dates) {
  const HINDI_MONTHS = [
    'जनवरी','फरवरी','मार्च','अप्रैल','मई','जून',
    'जुलाई','अगस्त','सितम्बर','अक्टूबर','नवम्बर','दिसम्बर'
  ];
  const groups = [];
  dates.forEach(d => {
    const key = `${d.getMonth()}-${d.getFullYear()}`;
    const last = groups[groups.length - 1];
    if (last && last.key === key) {
      last.dates.push(d);
    } else {
      groups.push({
        key,
        label: HINDI_MONTHS[d.getMonth()] + ' ' + d.getFullYear(),
        dates: [d]
      });
    }
  });
  return groups;
}
```

```javascript
    height:80px;writing-mode:vertical-rl;transform:rotate(180deg);`;
  monthRow += `<th rowspan="2" style="${SUM_HDR}">
    उपस्थिति पत्रक अवधि में लिए गए CL का योग</th>`;
  monthRow += `<th rowspan="2" style="${SUM_HDR}">

  monthRow += `<th rowspan="2" style="${SUM_HDR}">
    अब तक कुल ली गई CL का योग</th>`;
  monthRow += '</tr>';
  html += monthRow;

  // ROW 7: Date number headers (one per date)
  html += '<tr>';
  dates.forEach(function(d) {
    html += `<th style="${B}${CENTER}${VMID}font-size:7pt;width:18px;
      padding:1px;">${d.getDate()}</th>`;
  });
  html += '</tr>';

  // DATA ROWS — one per staff member
  var CELL = `${B}${CENTER}${VMID}`;
  var VERT_CELL = `${CELL}font-size:6pt;writing-mode:vertical-rl;
    transform:rotate(180deg);width:18px;padding:1px;overflow:hidden;
    white-space:nowrap;`;

  attState.staff.forEach(function(s, idx) {
    var currentCL = 0;
    var dateCells = '';
    dates.forEach(function(d) {
      var ds = d.toISOString().split('T')[0];
      var val = (s.days[ds] || 'उपस्थित');
      if (val === 'CL') currentCL++;

      // Cell background by value
      var bg = '';
      if (val === 'CL') bg = 'background:#fffde7;';
      else if (val === 'Day Off') bg = 'background:#e3f2fd;';
      else if (val === 'Willful Absence') bg = 'background:#ffebee;';
      else if (val === 'Onduty') bg = 'background:#e8f5e9;';

      dateCells += `<td style="${VERT_CELL}${bg}">${esc(val)}</td>`;
    });

    var prevCL  = parseInt(s.prevLeaves) || 0;
    var totalCL = currentCL + prevCL;

    html += '<tr>';
    html += `<td style="${CELL}font-size:8pt;">${idx + 1}</td>`;
    // Name + designation stacked
    html += `<td style="${B}${VMID}font-size:8pt;padding:2px 4px;">
      <div style="font-weight:bold;">${esc(s.name)}</div>
      <div style="font-size:7pt;color:#444;">${esc(s.designation)}</div>
    </td>`;
    html += dateCells;
    html += `<td style="${CELL}font-size:8pt;font-weight:bold;"
      id="att-doc-n1-${idx}">${currentCL}</td>`;
    html += `<td style="${CELL}font-size:8pt;"
      id="att-doc-n2-${idx}">${prevCL}</td>`;
    html += `<td style="${CELL}font-size:8pt;font-weight:bold;"
      id="att-doc-n3-${idx}">${totalCL}</td>`;
    html += '</tr>';
  });

  // NOTE ROW (if any)
  if (attState.note && attState.note.trim()) {
    html += `<tr><td colspan="${totalCols}" style="${B}text-align:left;
      font-size:8pt;padding:4px;">
      <strong>नोट :</strong> ${esc(attState.note.trim())}
    </td></tr>`;
  }

  // CERTIFICATION TEXT ROW
  html += `<tr><td colspan="${totalCols}" style="${B}${CENTER}${VMID}${BOLD}
    font-size:8pt;padding:8px;">
    प्रमाणित किया जाता है कि उपस्थिति पत्रक का मिलान उपस्थिति पंजिका से 
    कर लिया गया है, साथ ही कोई भी कार्मिक बिना सक्षम स्तर से अवकाश 
    स्वीकृत कराए उपस्थिति पत्रक में उल्लिखित अवधि के दौरान अनुपस्थित 
    नहीं रहा है।
  </td></tr>`;

  // SIGNATURE ROW
  html += `<tr><td colspan="${totalCols}" style="${B}text-align:right;
    font-size:8pt;padding:10px 12px 6px;">
    <div style="display:inline-block;text-align:center;min-width:180px;">
      <div style="margin-bottom:28px;">हस्ताक्षर प्रभारी</div>
      <div style="border-top:1px solid #000;padding-top:4px;font-size:7pt;">
        ${esc(attState.officeName || '')}
      </div>
    </div>
  </td></tr>`;

  tableEl.innerHTML = html;

  // Show preview body
  var previewBody = document.querySelector('.att-preview-body');
  if (previewBody && dates.length > 0) {
    previewBody.style.display = 'block';
  }
}
```

**4.8 — Validation (updated for new fields)**

```javascript
function validateAttForm() {
  const errs = [];
  if (!attState.officeName.trim()) 
    errs.push('कार्यालय का नाम अनिवार्य है।');
  if (!attState.periodFrom || !attState.periodTo) 
    errs.push('उपस्थिति अवधि चुनना आवश्यक है।');
  else if (new Date(attState.periodFrom) > new Date(attState.periodTo))
    errs.push('प्रारंभ तिथि समाप्ति तिथि से पहले होनी चाहिए।');
  if (attState.staff.length === 0)
    errs.push('कृपया कम से कम एक कार्मिक जोड़ें।');
  // Validate previous CL
  attState.staff.forEach((s, i) => {
    if (parseInt(s.prevLeaves) > 15)
      errs.push(`कार्मिक ${i+1} की पूर्व CL 15 से अधिक नहीं हो सकती।`);
  });
  return errs;
}
```
