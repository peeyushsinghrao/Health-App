STEP 2 — Replace the entire PDF export function with this exact code:

async function downloadPDF() {
  // 1. Get the attendance sheet wrapper element
  const element = document.getElementById('printArea');
  // Change 'printArea' to whatever ID wraps the attendance sheet

  // 2. Wait for all fonts to fully load
  await document.fonts.ready;
  await new Promise(r => setTimeout(r, 1500));

  // 3. Capture the live browser view as image
  const canvas = await html2canvas(element, {
    scale: 2,
    useCORS: true,
    allowTaint: true,
    backgroundColor: '#ffffff',
    scrollX: 0,
    scrollY: 0,
    windowWidth: element.scrollWidth,
    windowHeight: element.scrollHeight
  });

  // 4. Convert canvas to image and build PDF
  const imgData = canvas.toDataURL('image/png');
  const { jsPDF } = window.jspdf;

  const pdf = new jsPDF({
    orientation: 'landscape',
    unit: 'px',
    format: [canvas.width / 2, canvas.height / 2]
  });

  pdf.addImage(imgData, 'PNG', 0, 0,
    canvas.width / 2, canvas.height / 2);

  pdf.save('Upasthiti_Patrak.pdf');
}

STEP 3 — Wire the download button:
<button onclick="downloadPDF()">PDF डाउनलोड करें</button>

WHY: This takes a screenshot of the exact live browser view.
What you see in preview = what gets saved as PDF.
Fonts, layout, and everything are preserved perfectly.
